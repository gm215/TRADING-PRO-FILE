rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function staffPath() { return /databases/$(database)/documents/staff/$(request.auth.uid); }
    function isStaffOf(owner) { return signedIn() && exists(staffPath()) && get(staffPath()).data.owner == owner && get(staffPath()).data.active == true; }
    function roleIn(owner, roles) { return isStaffOf(owner) && get(staffPath()).data.role in roles; }
    function isOwner(owner) { return signedIn() && request.auth.uid == owner; }
    function member(owner) { return isOwner(owner) || isStaffOf(owner); }
    function office(owner) { return isOwner(owner) || roleIn(owner, ['accountant']); }
    function sales(owner) { return isOwner(owner) || roleIn(owner, ['accountant', 'sales']); }

    match /staff/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || resource.data.owner == request.auth.uid);
      allow create: if signedIn() && request.auth.uid != uid && request.resource.data.owner == request.auth.uid;
      allow update: if signedIn() && resource.data.owner == request.auth.uid && request.resource.data.owner == request.auth.uid;
      allow delete: if signedIn() && resource.data.owner == request.auth.uid;
    }
    match /users/{owner} {
      match /shipments/{id} { allow read, create, update: if member(owner); allow delete: if isOwner(owner); }
      match /shipsale/{id} { allow read, create, update: if sales(owner); allow delete: if isOwner(owner); }
      match /shipfin/{id} { allow read, create, update: if office(owner); allow delete: if isOwner(owner); }
      match /oldinv/{id} { allow read, create, update: if office(owner); allow delete: if isOwner(owner); }
      match /quotes/{id} { allow read, create, update: if sales(owner); allow delete: if isOwner(owner); }
      match /customers/{id} { allow read: if member(owner); allow create, update: if sales(owner); allow delete: if isOwner(owner); }
      match /photos/{id} { allow read, create, update: if member(owner); allow delete: if isOwner(owner); }
      match /settings/{id} { allow read: if member(owner); allow write: if isOwner(owner); }
      match /counters/{id} { allow read, write: if member(owner); }
      match /members/{uid} { allow read, write: if isOwner(owner); }
      match /activity/{id} { allow create: if member(owner) && request.resource.data.uid == request.auth.uid; allow read, delete: if isOwner(owner); }
    }
  }
}
